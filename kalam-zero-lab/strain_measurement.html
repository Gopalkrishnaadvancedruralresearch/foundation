Arduino Cantilever Strain Sketch 
#include "HX711.h"

// Required Library: "HX711 by Bogdan Necula"
// In Arduino IDE: Sketch -> Include Library -> Manage Libraries...

const int LOADCELL_DOUT_PIN = 2; // Data Pin (DT)
const int LOADCELL_SCK_PIN  = 3; // Clock Pin (SCK)

HX711 scale;

// Calibration constant for 500 Ohm Quarter Bridge Cantilever setup
// (GF = 2.00, Gain = 128, Bridge Excitation = 4.3V)
// Directly maps raw counts to microstrain (ue)
float microstrain_scale_factor = 214.75; 

void setup() {
  Serial.begin(115200);
  while (!Serial) { delay(10); }

  Serial.println("=================================================");
  Serial.println(" GARRF Lab - BRFX-500 Cantilever Strain Gauge ");
  Serial.println("=================================================");

  scale.begin(LOADCELL_DOUT_PIN, LOADCELL_SCK_PIN);

  if (scale.is_ready()) {
    scale.set_scale();
    Serial.println("Zeroing unstrained baseline (0 g load on beam)...");
    delay(1500);
    scale.tare(); // Zero offset
    scale.set_scale(microstrain_scale_factor);
    Serial.println("Baseline Zeroed. Outputting Microstrain (ue)...");
  } else {
    Serial.println("Error: HX711 not detected. Check breadboard wiring!");
  }
}

void loop() {
  if (scale.is_ready()) {
    // Read 5-sample averaged calibrated microstrain (ue)
    float microstrain = scale.get_units(5);
    long raw_counts   = scale.read_average(5);

    Serial.print("Raw ADC: ");
    Serial.print(raw_counts);
    Serial.print(" | Strain: ");
    Serial.print(microstrain, 1);
    Serial.println(" ue");
  } else {
    Serial.println("HX711 busy...");
  }
  delay(150);
}

Laboratory Guide: 4-Node Top-View Breadboard Guide

Understanding the 4 Connection Points:
A Wheatstone bridge consists of exactly four electrical junction points (nodes). On a breadboard, each node simply corresponds to a single row of connected holes where all component leads meet:

The 4 Breadboard Junction Points

  • Point 1 - Top Node (Red Dot / E+): Red wire from HX711 E+ meets the first leg of R1 and the first leg of R2.
  • Point 2 - Left Node (Green Dot / A+): Second leg of R1 meets the green wire from HX711 A+ and Lead A (Red) of the BRFX-500 strain gauge.
  • Point 3 - Right Node (Blue Dot / A-): Second leg of R2 meets the blue wire from HX711 A- and the first leg of R4.
  • Point 4 - Bottom Node (Black Dot / E- GND): Lead B (Black) of the BRFX-500 meets the second leg of R4 and the black wire from HX711 E- (GND).

Cantilever Beam Bending Mechanics

Applied Force: P = mass (kg) × 9.80665 m/s²
Bending Moment at Gauge: M = P × (L - x_gauge)
Surface Strain: ε = 6 × M / (E × b × t²)
Microstrain (με) = ε × 1,000,000

Gauge Resistance Shift: ΔR = R0 × GF × ε
Differential Bridge Voltage: V_diff = Vex × [ R_sensor / (R1 + R_sensor) - R4 / (R2 + R4) ]
At 0 g load (ΔR = 0): V_diff = 0.000 mV (Balanced)
